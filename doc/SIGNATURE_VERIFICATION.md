# Signature Verification Examples

**DEVELOPMENT TESTED** - Verified with real JavaCard hardware signatures

This document shows how to verify ECDSA signatures generated by the Mokapot applet.

## Overview

The Mokapot applet returns ECDSA signatures in DER format. To verify these signatures, you need to:

1. Parse the DER-encoded signature to extract `r` and `s` values
2. Use the public key from the applet 
3. Verify against the original message hash using ECDSA-P256-SHA256

## DER Signature Format

The applet returns signatures in this format:
```
30 [total-length]     -- SEQUENCE tag
   02 [r-length] [r]  -- INTEGER r
   02 [s-length] [s]  -- INTEGER s
```

## Python Example with cryptography library

```python
#!/usr/bin/env python3
import hashlib
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.primitives.asymmetric import ec
from cryptography.hazmat.primitives.asymmetric.utils import decode_dss_signature

def verify_mokapot_signature(message, signature_der, public_key_bytes):
    """
    Verify a signature from the Mokapot applet.
    
    Args:
        message (bytes): Original message that was signed
        signature_der (bytes): DER-encoded signature from applet
        public_key_bytes (bytes): 65-byte uncompressed public key from applet
        
    Returns:
        bool: True if signature is valid
    """
    
    # Parse the uncompressed public key (65 bytes: 0x04 + 32-byte X + 32-byte Y)
    if len(public_key_bytes) != 65 or public_key_bytes[0] != 0x04:
        raise ValueError("Invalid public key format")
    
    x = int.from_bytes(public_key_bytes[1:33], 'big')
    y = int.from_bytes(public_key_bytes[33:65], 'big')
    
    # Create cryptography public key object
    public_key = ec.EllipticCurvePublicNumbers(x, y, ec.SECP256R1()).public_key()
    
    # Hash the message (same as what was sent to the card)
    message_hash = hashlib.sha256(message).digest()
    
    try:
        # Verify the signature
        public_key.verify(
            signature_der,
            message_hash,
            ec.ECDSA(hashes.SHA256())
        )
        return True
    except:
        return False
```

## JavaScript Example with Web Crypto API

```javascript
async function verifyMokapotSignature(message, signatureDer, publicKeyBytes) {
    // Parse uncompressed public key
    if (publicKeyBytes.length !== 65 || publicKeyBytes[0] !== 0x04) {
        throw new Error("Invalid public key format");
    }
    
    const x = publicKeyBytes.slice(1, 33);
    const y = publicKeyBytes.slice(33, 65);
    
    // Import the public key
    const publicKey = await crypto.subtle.importKey(
        'raw',
        new Uint8Array([0x04, ...x, ...y]),
        { name: 'ECDSA', namedCurve: 'P-256' },
        false,
        ['verify']
    );
    
    // Hash the message
    const messageHash = await crypto.subtle.digest('SHA-256', message);
    
    // Parse DER signature to extract r,s values
    const { r, s } = parseDerSignature(signatureDer);
    
    // Convert to Web Crypto format (r || s, both 32 bytes)
    const signature = new Uint8Array(64);
    signature.set(r.slice(-32), 0);  // Pad r to 32 bytes
    signature.set(s.slice(-32), 32); // Pad s to 32 bytes
    
    // Verify
    return await crypto.subtle.verify(
        { name: 'ECDSA', hash: 'SHA-256' },
        publicKey,
        signature,
        messageHash
    );
}

function parseDerSignature(derBytes) {
    // Simple DER parser for ECDSA signatures
    let offset = 0;
    
    if (derBytes[offset++] !== 0x30) throw new Error("Invalid DER sequence");
    const totalLength = derBytes[offset++];
    
    if (derBytes[offset++] !== 0x02) throw new Error("Invalid DER integer");
    const rLength = derBytes[offset++];
    const r = derBytes.slice(offset, offset + rLength);
    offset += rLength;
    
    if (derBytes[offset++] !== 0x02) throw new Error("Invalid DER integer");
    const sLength = derBytes[offset++];
    const s = derBytes.slice(offset, offset + sLength);
    
    return { r, s };
}
```

## OpenSSL Command Line

You can also verify signatures using OpenSSL:

```bash
# 1. Save the public key in PEM format
echo "-----BEGIN PUBLIC KEY-----" > pubkey.pem
echo -n "04" | xxd -r -p > pubkey_raw.bin  # Start with 0x04
cat card_public_key.bin >> pubkey_raw.bin   # Add the 64 bytes from card
openssl ec -inform raw -pubin -in pubkey_raw.bin -outform PEM >> pubkey.pem
echo "-----END PUBLIC KEY-----" >> pubkey.pem

# 2. Create the message hash  
echo -n "Hello, EspreSSHo!" | openssl dgst -sha256 -binary > message.hash

# 3. Save the DER signature from the card
echo "3044022012345678..." | xxd -r -p > signature.der

# 4. Verify
openssl dgst -sha256 -verify pubkey.pem -signature signature.der message.hash
```

## Go Example

```go
package main

import (
    "crypto/ecdsa"
    "crypto/elliptic"
    "crypto/sha256"
    "encoding/asn1"
    "fmt"
    "math/big"
)

type ECDSASignature struct {
    R, S *big.Int
}

func verifyMokapotSignature(message, signatureDER, publicKeyBytes []byte) bool {
    // Parse uncompressed public key
    if len(publicKeyBytes) != 65 || publicKeyBytes[0] != 0x04 {
        return false
    }
    
    // Extract coordinates
    x := new(big.Int).SetBytes(publicKeyBytes[1:33])
    y := new(big.Int).SetBytes(publicKeyBytes[33:65])
    
    // Create public key
    pubKey := &ecdsa.PublicKey{
        Curve: elliptic.P256(),
        X:     x,
        Y:     y,
    }
    
    // Parse DER signature
    var sig ECDSASignature
    if _, err := asn1.Unmarshal(signatureDER, &sig); err != nil {
        return false
    }
    
    // Hash message
    hash := sha256.Sum256(message)
    
    // Verify
    return ecdsa.Verify(pubKey, hash[:], sig.R, sig.S)
}
```
